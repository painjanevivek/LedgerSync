export type ClientOptions = {
  baseUrl: string;
  fetch?: typeof globalThis.fetch;
  token: string;
  /** Own-principal clients omit this. Delegating backends mint one assertion per call. */
  actorAssertion?: string | (() => string | Promise<string>);
  timeoutMilliseconds?: number;
};

export function createClient(options: ClientOptions) {
  const request = options.fetch ?? globalThis.fetch;
  const base = new URL(options.baseUrl);
  const timeout = options.timeoutMilliseconds ?? 10_000;
  if (!['http:', 'https:'].includes(base.protocol) || base.username || base.password || base.search || base.hash || !Number.isSafeInteger(timeout) || timeout < 100 || timeout > 30_000) throw new Error('Invalid bounded client configuration');
  if (base.protocol === 'http:' && !['127.0.0.1', 'localhost', '[::1]'].includes(base.hostname)) throw new Error('Plain HTTP is limited to an explicit loopback test server');
  const prefix = base.pathname.replace(/\/$/, '') + '/';
  const root = new URL(base.href); root.pathname = prefix;
  return {
    operations,
    async call(operation: Operation, path: string, init: RequestInit = {}) {
      if (!path.startsWith('/') || path.startsWith('//') || path.includes('\\')) throw new Error('An API-relative operation path is required');
      const target = new URL(path.slice(1), root);
      if (target.origin !== base.origin || !target.pathname.startsWith(prefix) || target.hash) throw new Error('Operation path escaped the configured API base');
      const headers = new Headers(init.headers);
      headers.set('Authorization', `Bearer ${options.token}`);
      const assertion = typeof options.actorAssertion === 'function' ? await options.actorAssertion() : options.actorAssertion;
      if (assertion) headers.set('X-LedgerSync-Actor-Assertion', assertion);
      else headers.delete('X-LedgerSync-Actor-Assertion');
      const deadline = AbortSignal.timeout(timeout);
      return request(target, { ...init, method: operation.method, headers, redirect: 'error', signal: init.signal ? AbortSignal.any([init.signal, deadline]) : deadline });
    },
  };
}
